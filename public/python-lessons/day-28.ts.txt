def execute_network_ping_diagnostic(host: str, timeout_sec: int = 5) -> dict:
    \"\"\"
    Executes a system network ping command against a host server,
    safely handling timeouts, non-zero exit codes, and platform differences.
    \"\"\"
    print(f"\n--- ADVANCED: Executing Ping Diagnostic for [{host}] ---")

    # Construct platform-specific argument list securely (NO shell=True!)
    if sys.platform == "win32":
        ping_cmd = ["ping", "-n", "1", "-w", str(timeout_sec * 1000), host]
    else:
        ping_cmd = ["ping", "-c", "1", "-W", str(timeout_sec), host]

    try:
        # Execute process with strict timeout guard
        completed_proc = subprocess.run(
            ping_cmd,
            capture_output=True,
            text=True,
            timeout=timeout_sec
        )

        is_successful = completed_proc.returncode == 0

        return {
            "host": host,
            "success": is_successful,
            "exit_code": completed_proc.returncode,
            "output": completed_proc.stdout.strip() if is_successful else completed_proc.stderr.strip()
        }
    except subprocess.TimeoutExpired:
        return {"host": host, "success": False, "error": f"TimeoutExpired: Host failed to respond within {timeout_sec}s."}
    except FileNotFoundError:
        return {"host": host, "success": False, "error": "FileNotFoundError: 'ping' binary not found on host OS."}

<b>Advanced File Copying, Moving, Archiving, and Disk Space Inspection</b><br/>
While `pathlib` and `os` handle basic single-file creation and deletion, complex file system management—such as copying entire directory trees, moving files across file systems, creating ZIP archives, or checking available disk space—requires the high-level <b><code>shutil</code></b> (Shell Utilities) module.<br/><br/>

<b>1. High-Level File Copying and Moving:</b>
<ul>
  <li><code>shutil.copy(src, dst)</code>: Copies a single file from `src` to `dst`, preserving file permissions.</li>

# Advanced / Production-Style: Safe Workspace Cleanup Engine
# ------------------------------------------------------------------------------

def safe_purge_directory_contents(target_folder: Path) -> dict:
    \"\"\"
    Safely purges all contents inside a target folder using shutil.rmtree
    while preserving the root folder itself.
    \"\"\"
    print(f"\n--- ADVANCED: Purging Directory [{target_folder}] ---")

    if not target_folder.exists() or not target_folder.is_dir():
        return {"success": False, "error": "Invalid target directory."}

    purged_count = 0
    for child in target_folder.iterdir():
        if child.is_file() or child.is_symlink():
            child.unlink()
            purged_count += 1
