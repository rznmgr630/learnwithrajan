<b>1. The Attribute Lookup Order:</b><br/>
When accessing <code>obj.attr</code>, Python evaluates locations in this exact sequence:
<ol>

<b>Reusable Attribute Access Control and the Descriptor Protocol</b><br/>
In Python, **Descriptors** are the secret weapon behind properties, bound methods, <code>@classmethod</code>, <code>@staticmethod</code>, and ORM fields (like Django or SQLAlchemy models). A Descriptor is a Python object that customizes how attribute access is handled across instances by implementing the **Descriptor Protocol**.<br/><br/>

<b>1. The Descriptor Protocol Dunder Methods:</b><br/>
An object becomes a Descriptor when its class implements one or more of these methods:
<ul>

<ul>

class CustomPropertyDescriptor:
    """Simplified implementation showing how @property works internally."""
    def __init__(self, fget=None, fset=None):
        self.fget = fget
        self.fset = fset

    def __get__(self, instance, owner=None):
        if instance is None:
            return self
        if self.fget is None:
            raise AttributeError("Unreadable attribute")
        return self.fget(instance)

    def __set__(self, instance, value):
        if self.fset is None:
            raise AttributeError("Can't set attribute")
        self.fset(instance, value)


class DeliveryManifest:
    def __init__(self, base_fee: float):
        self._base_fee = base_fee

    # Using custom property descriptor
    def _get_fee(self):
        return self._base_fee * 1.10 # Include 10% surcharge

    def _set_fee(self, value):
        if value <= 0:
            raise ValueError("Fee must be positive")
        self._base_fee = value

    fee = CustomPropertyDescriptor(fget=_get_fee, fset=_set_fee)

manifest = DeliveryManifest(100.0)

print("--- BEGINNER: Custom Property Descriptor ---")
print("Calculated Fee ($):", manifest.fee) # 110.0

print("Unbound Class Function :", unbound_func) # <function ParcelTracker.track>

# Accessing function from Instance invokes function.__get__ to bind 'self'!
bound_method = tracker.track
print("Bound Instance Method  :", bound_method) # <bound method ParcelTracker.track of ...>

# Executing bound method (No need to pass 'tracker' manually!)
print("Invoking Bound Method  :", bound_method())

class ModelFieldDescriptor:
    """Base descriptor for Mini-ORM Model fields."""
    def __init__(self, data_type, primary_key: bool = False):
        self.data_type = data_type
        self.primary_key = primary_key

    def __set_name__(self, owner, name):
        self.column_name = name
        self.private_name = "_" + name

    def __get__(self, instance, owner=None):
        if instance is None:
            return self
        return getattr(instance, self.private_name, None)

    def __set__(self, instance, value):
        if value is not None and not isinstance(value, self.data_type):
            raise TypeError(f"Field '{self.column_name}' expects {self.data_type.__name__}, got {type(value).__name__}")

        # ORM Dirty State Tracking
        old_val = getattr(instance, self.private_name, None)
        if old_val != value:
            print(f"  [ORM DIRTY TRACKER] Field '{self.column_name}' modified: '{old_val}' -> '{value}'")
            instance._is_dirty = True

        setattr(instance, self.private_name, value)


class MiniORMModel:
    """Base ORM model class."""
    def __init__(self, **kwargs):
        self._is_dirty = False
        for k, v in kwargs.items():
            setattr(self, k, v)
        self._is_dirty = False # Reset dirty flag after initial load

    def save(self):
        if self._is_dirty:
            print(f"[ORM SQL EXECUTE] Syncing dirty model changes to PostgreSQL database...")
            self._is_dirty = False
        else:
            print("[ORM SQL NOTICE] No changes detected. Database write skipped.")
