def ingest_regional_scan_event(
    raw_scan_time: str,
    hub_iana_zone: str
) -> dict:
    \"\"\"
    Ingests a naive local scan timestamp string, attaches regional IANA timezone,
    and normalizes the event to UTC for database storage.
    \"\"\"
    print(f"\n--- ADVANCED: Ingesting Event from [{hub_iana_zone}] ---")

    # Parse naive timestamp
    naive_dt = datetime.fromisoformat(raw_scan_time)

    # Attach regional IANA timezone (replace tzinfo)
    regional_tz = ZoneInfo(hub_iana_zone)
    regional_aware_dt = naive_dt.replace(tzinfo=regional_tz)

    # Normalize to UTC for database storage (.astimezone)
    utc_normalized_dt = regional_aware_dt.astimezone(timezone.utc)

    return {
        "local_input": raw_scan_time,
        "regional_aware_iso": regional_aware_dt.isoformat(),
        "utc_storage_iso": utc_normalized_dt.isoformat(),
