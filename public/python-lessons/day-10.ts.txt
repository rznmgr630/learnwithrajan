def process_complex_telemetry_batch(telemetry_batches: list[dict]) -> list[dict]:
    \"\"\"
    Demonstrates breaking complex data transformations into clean, 
    maintainable helper functions and standard loops rather than unreadable single-liners.
    \"\"\"
    print("\n--- ADVANCED: Maintainable Telemetry Pipeline ---")
    
    def is_critical_error(record: dict) -> bool:
        return record.get("status") == "FAILED" and record.get("retry_count", 0) >= 3

    # Clean, self-documenting processing loop
    critical_alerts = []
    
    for batch in telemetry_batches:
        hub_name = batch.get("hub_code", "UNKNOWN")
        records = batch.get("telemetry_logs", [])
        
        # Single-level clean list comprehension for filtered batch logs
        flagged_logs = [r for r in records if is_critical_error(r)]
        
        for log in flagged_logs:
            critical_alerts.append({
