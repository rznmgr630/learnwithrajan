import json
import os
from typing import List, Dict, Any
from exceptions import StorageError

class JSONStorage:
    def __init__(self, filepath: str = "expenses.json"):
        self.filepath = filepath

    def load_data(self) -> List[Dict[str, Any]]:
        """Load data from JSON file. Returns an empty list if file doesn't exist."""
        if not os.path.exists(self.filepath):
            return []
        
        try:
            with open(self.filepath, "r", encoding="utf-8") as file:
                return json.load(file)
        except (json.JSONDecodeError, OSError) as e:
            raise StorageError(f"Failed to read storage file '{self.filepath}': {e}")

    def save_data(self, records: List[Dict[str, Any]]) -> None:
        """Atomically persist records to JSON storage."""
        try:
            with open(self.filepath, "w", encoding="utf-8") as file:
                json.dump(records, file, indent=2)
        except OSError as e:
            raise StorageError(f"Failed to write to storage file '{self.filepath}': {e}")
`.trim()

from typing import List, Dict, Any
from storage import JSONStorage
from exceptions import InvalidTransactionError

class ExpenseTracker:
    VALID_TYPES = {"income", "expense"}

    def __init__(self, storage: JSONStorage):
        self.storage = storage
        self.records: List[Dict[str, Any]] = self.storage.load_data()

    def add_transaction(self, t_type: str, amount: float, category: str, description: str = "") -> Dict[str, Any]:
        """Validate and add a new income or expense transaction."""
        t_type = t_type.lower().strip()
        if t_type not in self.VALID_TYPES:
            raise InvalidTransactionError(f"Type must be one of {self.VALID_TYPES}")
        
        if amount <= 0:
            raise InvalidTransactionError("Amount must be greater than zero.")

        if not category.strip():
            raise InvalidTransactionError("Category cannot be empty.")

        next_id = max([r.get("id", 0) for r in self.records], default=0) + 1
        
        record = {
            "id": next_id,
            "type": t_type,
            "amount": round(amount, 2),
            "category": category.strip().title(),
            "description": description.strip()
        }
        
        self.records.append(record)
        self.storage.save_data(self.records)
        return record

    def get_summary(self) -> Dict[str, float]:
        """Compute total income, total expenses, and net balance."""
        income = sum(r["amount"] for r in self.records if r["type"] == "income")
        expenses = sum(r["amount"] for r in self.records if r["type"] == "expense")
        return {
            "income": round(income, 2),
            "expenses": round(expenses, 2),
            "balance": round(income - expenses, 2)
        }
`.trim()
