def generate_model_class_from_schema(class_name: str, schema_fields: dict) -> type:
    """Dynamically builds a custom model class from a dictionary field schema."""

    def __init__(self, **kwargs):
        for field_name, default_val in schema_fields.items():
            # Assign kwarg or fallback default
            setattr(self, field_name, kwargs.get(field_name, default_val))

    def to_dict(self):
        return {field: getattr(self, field) for field in schema_fields}

    # Construct class dynamically
    generated_class = type(
        class_name,
        (object,),
        {
            "__init__": __init__,
            "to_dict": to_dict,
            "_schema": schema_fields
        }
    )
    return generated_class

def inspect_class_endpoint_signatures(target_class: type) -> dict:
    \"\"\"
    Uses the inspect module to extract parameter signatures and docstrings
    from all public methods on a class automatically.
    \"\"\"
    print(f"\n--- ADVANCED: Inspecting Class [{target_class.__name__}] ---")

    endpoint_catalog = {}

    for attr_name in dir(target_class):
        if attr_name.startswith("_"):
            continue

        attr_val = getattr(target_class, attr_name)

        # Verify attribute is a callable function/method
        if callable(attr_val):
            sig = inspect.signature(attr_val)
            doc = inspect.getdoc(attr_val) or "No docstring provided."

            endpoint_catalog[attr_name] = {
                "parameters": [p for p in sig.parameters.keys() if p != "self"],
                "signature": str(sig),
                "docstring": doc
            }

    return endpoint_catalog

# Run class inspector
